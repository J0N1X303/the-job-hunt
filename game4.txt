
  function drawLauncher(now){
    const x=W/2,y=H-112;roundRect(x-36,y-25,72,50,14,'#101c2f','rgba(255,255,255,.12)');
    ctx.fillStyle='#f7f9fc';ctx.font='900 25px system-ui';ctx.textAlign='center';ctx.fillText('📄',x,y+8);ctx.fillStyle='#6f7f98';ctx.font='800 9px system-ui';ctx.fillText('YOUR CV',x,y+38);
    if(!firstShot){
      const a=.55+.35*Math.sin(now/180);ctx.globalAlpha=a;ctx.strokeStyle='#55d8ff';ctx.lineWidth=2;ctx.setLineDash([6,7]);
      ctx.beginPath();ctx.moveTo(x,y-30);ctx.lineTo(target.x+target.w/2,target.y+target.h+10);ctx.stroke();ctx.setLineDash([]);
      ctx.fillStyle='#55d8ff';ctx.font='900 13px system-ui';ctx.fillText('TAP ANYWHERE TO SEND CV',W/2,H-175);ctx.globalAlpha=1;
    }
  }

  function drawPickups(now){
    pickups.forEach(p=>{
      const bob=Math.sin(now/260+p.phase)*3;const x=p.x,y=p.y+bob;
      ctx.save();ctx.shadowColor='rgba(85,216,255,.35)';ctx.shadowBlur=14;roundRect(x,y,p.w,p.h,14,'rgba(12,37,47,.95)','rgba(85,216,255,.45)');ctx.restore();
      ctx.textAlign='left';ctx.fillStyle='#f7f9fc';ctx.font='900 16px system-ui';ctx.fillText(p.def.icon,x+10,y+22);
      ctx.font='800 10px system-ui';ctx.fillStyle='#cfefff';ctx.fillText(p.def.label,x+38,y+20);
      ctx.font='800 9px system-ui';ctx.fillStyle='#55d8ff';ctx.fillText('TAP TO COLLECT',x+38,y+38);
    });
  }

  function drawShots(){
    shots.forEach(s=>{ctx.save();s.trail.forEach((p,i)=>{ctx.globalAlpha=(i+1)/s.trail.length*.18;roundRect(p.x-10,p.y-6,20,12,3,'#55d8ff');});ctx.globalAlpha=1;
      ctx.translate(s.x,s.y);ctx.rotate(Math.atan2(s.vy,s.vx)+Math.PI/2);roundRect(-14,-9,28,18,4,s.auto?'#c9ff5d':'#f4f7fb');ctx.fillStyle='#102033';ctx.fillRect(-9,-4,14,2);ctx.fillRect(-9,1,10,2);ctx.restore();});
  }

  function drawParticles(){particles.forEach(p=>{ctx.globalAlpha=Math.max(0,p.life);ctx.fillStyle=p.color;ctx.beginPath();ctx.arc(p.x,p.y,p.size,0,Math.PI*2);ctx.fill();ctx.globalAlpha=1;});}

  function drawSuper(now){
    const a=.5+.25*Math.sin(now/80);ctx.strokeStyle=`rgba(201,255,93,${a})`;ctx.lineWidth=4;ctx.strokeRect(3,3,W-6,H-6);
    ctx.fillStyle='rgba(201,255,93,.96)';ctx.font='900 13px system-ui';ctx.textAlign='center';ctx.fillText('☎️ WARM INTRO — BARRIERS OPEN',W/2,H*.25);
  }

  function drawLevelBanner(now){
    const remaining=Math.max(0,levelBannerUntil-now);const alpha=Math.min(1,remaining/350,(1500-remaining)/250);ctx.globalAlpha=Math.max(0,alpha);
    roundRect(W*.12,H*.22,W*.76,76,18,'rgba(8,16,29,.92)','rgba(201,255,93,.25)');ctx.textAlign='center';ctx.fillStyle='#c9ff5d';ctx.font='900 12px system-ui';ctx.fillText(`LEVEL ${level}`,W/2,H*.22+25);
    ctx.fillStyle='#f7f9fc';ctx.font='900 19px system-ui';ctx.fillText(levelNames[level-1],W/2,H*.22+50);ctx.globalAlpha=1;
  }

  function endGame(){
    if(state!=='play')return;state='over';cancelAnimationFrame(raf);hud.classList.add('hidden');gameover.classList.remove('hidden');
    stats.score=Math.round(stats.score+stats.interviews*110+stats.calls*180+stats.moves*60);
    bestScore=Math.max(bestScore,stats.score);bestInterviews=Math.max(bestInterviews,stats.interviews);bestLevel=Math.max(bestLevel,level);
    try{localStorage.setItem('jobhunt-best-score',bestScore);localStorage.setItem('jobhunt-best-interviews',bestInterviews);localStorage.setItem('jobhunt-best-level',bestLevel);}catch(_){}
    document.getElementById('bigScore').textContent=stats.score.toLocaleString();
    document.getElementById('sApps').textContent=stats.apps;document.getElementById('sInts').textContent=stats.interviews;document.getElementById('sRejects').textContent=stats.rejects;
    document.getElementById('sMoves').textContent=stats.moves;document.getElementById('sCalls').textContent=stats.calls;document.getElementById('sLevel').textContent=level;
    const result=stats.interviews>=4?'You made it through the hiring committee. Suspiciously competent.':stats.interviews>=2?'You reached the shortlist. Now try to beat it.':stats.interviews>=1?'You got the interview. Next run: go further.':'The ATS wins this round. The network might have ideas.';
    document.getElementById('resultLine').textContent=`${result} Best: ${bestInterviews} interviews · Level ${bestLevel}.`;
  }
