function drawGates(now){const superActive=now<superUntil;gates.forEach(g=>{const l=g.x-g.gap/2,r=g.x+g.gap/2;ctx.globalAlpha=superActive?.18:1;roundRect(0,g.y-g.h/2,Math.max(0,l),g.h,8,'#26344a');roundRect(r,g.y-g.h/2,Math.max(0,W-r),g.h,8,'#26344a');ctx.fillStyle=superActive?'#c9ff5d':'#ffc85c';ctx.font='900 10px system-ui';ctx.textAlign='left';ctx.fillText(g.label,10,g.y-18);ctx.strokeStyle=superActive?'rgba(201,255,93,.7)':'rgba(85,216,255,.34)';ctx.setLineDash([5,5]);ctx.beginPath();ctx.moveTo(l,g.y);ctx.lineTo(r,g.y);ctx.stroke();ctx.setLineDash([]);ctx.globalAlpha=1})}
function drawCV(){if(!cv||!cv.alive)return;ctx.save();ctx.translate(cv.x,cv.y);const rot=(controlX-cv.x)*.008;ctx.rotate(rot);roundRect(-15,-10,30,20,4,performance.now()<superUntil?'#c9ff5d':'#f5f7fb');ctx.fillStyle='#102033';ctx.fillRect(-9,-4,16,2);ctx.fillRect(-9,1,11,2);ctx.restore();if(cv.hits>0){ctx.textAlign='center';ctx.fillStyle='#ffc85c';ctx.font='800 9px system-ui';ctx.fillText(`${3-cv.hits} BOUNCE${3-cv.hits===1?'':'S'} LEFT`,cv.x,cv.y+27)}}
function drawPickups(now){pickups.forEach(p=>{const pulse=1+.06*Math.sin(now/220+p.pulse);ctx.save();ctx.translate(p.x,p.y);ctx.scale(pulse,pulse);ctx.beginPath();ctx.arc(0,0,p.r,0,Math.PI*2);ctx.fillStyle='rgba(201,255,93,.13)';ctx.fill();ctx.strokeStyle='rgba(201,255,93,.9)';ctx.lineWidth=2;ctx.stroke();ctx.textAlign='center';ctx.fillStyle='#f7f9fc';ctx.font='18px system-ui';ctx.fillText(p.m.icon,0,3);ctx.fillStyle='#c9ff5d';ctx.font='900 8px system-ui';ctx.fillText(p.m.short,0,17);ctx.restore()})}
function drawParticles(){particles.forEach(p=>{ctx.globalAlpha=Math.max(0,p.life);ctx.fillStyle=p.color;ctx.beginPath();ctx.arc(p.x,p.y,p.size,0,Math.PI*2);ctx.fill();ctx.globalAlpha=1})}
function drawHint(){if(state!=='play'||elapsed>4||stats.interviews>0)return;ctx.textAlign='center';ctx.fillStyle='rgba(247,249,252,.88)';ctx.font='900 13px system-ui';ctx.fillText('← DRAG TO STEER →',W/2,H-158);ctx.fillStyle='rgba(158,172,196,.9)';ctx.font='700 10px system-ui';ctx.fillText('Hit a barrier? Your CV bounces back.',W/2,H-140)}
function draw(now){ctx.clearRect(0,0,W,H);drawBackdrop();if(state==='play'){drawTarget();drawGates(now);drawPickups(now);drawCV();drawParticles();drawHint();if(now<superUntil){ctx.strokeStyle='rgba(201,255,93,.85)';ctx.lineWidth=4;ctx.strokeRect(3,3,W-6,H-6)}}}
