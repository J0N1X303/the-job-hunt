function update(dt,now){
 elapsed=(now-startTime)/1000;const superActive=now<superUntil;
 gates.forEach((g,i)=>{if(level>1)g.x=W/2+Math.sin(now/1000*g.speed+g.phase)*Math.max(8,W/2-g.gap/2-16)});
 if(cv&&cv.alive){
   cv.x+=(controlX-cv.x)*Math.min(1,dt*8.5);cv.x=Math.max(18,Math.min(W-18,cv.x));cv.y+=cv.vy*dt;
   if(cv.bounce>0){cv.bounce-=dt;if(cv.bounce<=0)cv.vy=superActive?-500:-330}
   if(!superActive&&cv.vy<0){for(let i=0;i<gates.length;i++){const g=gates[i];if(cv.cleared.has(i))continue;const l=g.x-g.gap/2,r=g.x+g.gap/2;const crossing=cv.y-cv.h/2<g.y+g.h/2&&cv.y+cv.h/2>g.y-g.h/2;const inside=cv.x-cv.w/2>l&&cv.x+cv.w/2<r;if(crossing){if(inside){cv.cleared.add(i);stats.score+=35;burst(cv.x,g.y,'#55d8ff',5)}else{bounce(g)}break}}}
   if(superActive)cv.vy=-500;
   for(const p of pickups){if(p.alive&&Math.hypot(cv.x-p.x,cv.y-p.y)<p.r+16)collectPickup(p)}
   if(cv.alive&&cv.y<target.y+target.h&&cv.x>target.x-12&&cv.x<target.x+target.w+12)interview();
   if(cv&&cv.alive&&cv.y>H-80&&cv.vy>0){cv.y=H-112;cv.vy=-330;cv.bounce=0}
 }
 pickups=pickups.filter(p=>p.alive);particles.forEach(p=>{p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=100*dt;p.life-=dt});particles=particles.filter(p=>p.life>0);stats.score+=dt*4;updateHud();if(elapsed>=duration)endGame()
}
