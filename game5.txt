  }

  function drawShots(){
    shots.forEach(s=>{
      ctx.save(); s.trail.forEach((p,i)=>{ctx.globalAlpha=(i+1)/s.trail.length*.18;roundRect(p.x-10,p.y-6,20,12,3,'#55d8ff');}); ctx.globalAlpha=1;
      ctx.translate(s.x,s.y); const ang=Math.atan2(s.vy,s.vx)+Math.PI/2;ctx.rotate(ang);
      roundRect(-14,-9,28,18,4,s.auto?'#c9ff5d':'#f4f7fb');
      ctx.fillStyle='#102033';ctx.fillRect(-9,-4,14,2);ctx.fillRect(-9,1,10,2); ctx.restore();
    });
  }

  function drawParticles(){ particles.forEach(p=>{ctx.globalAlpha=Math.max(0,p.life);ctx.fillStyle=p.color;ctx.beginPath();ctx.arc(p.x,p.y,p.size,0,Math.PI*2);ctx.fill();ctx.globalAlpha=1;}); }

  function drawSuper(now){

    ctx.fillStyle='rgba(201,255,93,.95)';ctx.font='900 13px system-ui';ctx.textAlign='center';ctx.fillText('☎️ HUMAN CONTACT — BARRIERS OPEN',W/2,H*.27);
  }

  function endGame(){
    if(state!=='play')return; state='over'; cancelAnimationFrame(raf); hud.classList.add('hidden'); gameover.classList.remove('hidden');
    stats.score=Math.round(stats.score+stats.interviews*120+stats.calls*160); best=Math.max(best,stats.score); try { sessionStorage.setItem('jobhunt-best',best); } catch(_) {}
    document.getElementById('bigScore').textContent=stats.score.toLocaleString();

  }

  async function share(){
    const text=shareText();
    try{ if(navigator.share){ await navigator.share({title:'The Job Hunt',text,url:location.protocol.startsWith('http')?location.href:undefined}); }
      else { await navigator.clipboard.writeText(text); showToast('Score copied — paste it into LinkedIn.'); }
    } catch(e) { if(e?.name!=='AbortError'){ try{await navigator.clipboard.writeText(text);showToast('Score copied.');}catch(_){showToast(text);} } }
  }
